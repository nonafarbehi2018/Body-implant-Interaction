color_to_identity <- function(colors_vec) {
  identity_map <- character(0)
  for (col in unique(colors_vec)) {
    labels_for_color <- names(colors_vec)[colors_vec == col]
    canonical <- sub("^C[0-9]+-", "", labels_for_color[1])
    identity_map[labels_for_color] <- canonical
  }
  identity_map
}

natural_cluster_order <- function(labels) {
  nums <- as.numeric(sub("^C([0-9]+).*", "\\1", labels))

}

get_imm_receiver <- function(cellchat) {
  idx <- which(grepl("IMM", levels(cellchat@idents)))
  if (length(idx) < 2) {
    fib_idx <- setdiff(which(grepl("FIB", levels(cellchat@idents))), idx)
    idx <- c(idx, head(fib_idx, 2 - length(idx)))
  }
  if (length(idx) < 2) idx <- seq_len(min(2, length(levels(cellchat@idents))))
  sort(idx)
}
build_cellchat <- function(seurat_obj, group_col = "cell_type_label_updated") {
  cellchat <- createCellChat(object = seurat_obj, group.by = group_col, assay = "RNA")
  cellchat@DB <- CellChatDB.use
  cellchat <- subsetData(cellchat)
  cellchat <- identifyOverExpressedGenes(cellchat)
  cellchat <- identifyOverExpressedInteractions(cellchat)
  cellchat <- computeCommunProb(cellchat, raw.use = TRUE)
  cellchat <- filterCommunication(cellchat, min.cells = 10)
  cellchat <- computeCommunProbPathway(cellchat)
  cellchat <- aggregateNet(cellchat)
  cellchat
}
build_raw_centr_matrix <- function(cellchat, pathways, pattern) {
  slot_names <- switch(pattern, outgoing = "outdeg", incoming = "indeg", all = c("outdeg", "indeg"))
  cell_types <- levels(cellchat@idents)
  mat <- matrix(0, nrow = length(pathways), ncol = length(cell_types), dimnames = list(pathways, cell_types))
  for (pw in pathways) if (pw %in% names(cellchat@netP$centr)) for (sn in slot_names) {
    v <- cellchat@netP$centr[[pw]][[sn]]; mat[pw, names(v)] <- mat[pw, names(v)] + v
  }
  mat
}

make_singler_heatmap <- function(obj, obj_name, label_col = "cell_type_label_updated", singler_col = "SingleR_label", out_path) {
  ordered_levels <- natural_cluster_order(unique(as.character(obj@meta.data[[label_col]])))
  tab <- table(Cluster = obj@meta.data[[label_col]], SingleR = obj@meta.data[[singler_col]])[ordered_levels, , drop = FALSE]
  count_mat <- as.matrix(unclass(tab)); pct_mat <- (count_mat / rowSums(count_mat)) * 100
  col_fun <- colorRamp2(c(0, 100), c("#cde2fb", "#104281"))
  cell_fun <- function(j, i, x, y, width, height, fill) {
    if (count_mat[i, j] > 0) {
      lum <- 0.299*col2rgb(fill)[1] + 0.587*col2rgb(fill)[2] + 0.114*col2rgb(fill)[3]
      grid.text(sprintf("%d\n(%.1f%%)", count_mat[i, j], pct_mat[i, j]), x, y,
                gp = gpar(fontsize = 7, col = if (lum > 140) "black" else "white"))
    }
  }
  ht <- Heatmap(pct_mat, name = "%", col = col_fun, cluster_rows = FALSE, cluster_columns = FALSE, cell_fun = cell_fun,
                show_row_names = TRUE, row_names_side = "right",
                column_title = paste0("SingleR % (updated labels): ", obj_name),
                column_title_gp = gpar(fontsize = 13, fontface = "bold"),
                row_names_gp = gpar(fontsize = 11, fontface = "bold"),
                column_names_gp = gpar(fontsize = 10, fontface = "bold"),
                heatmap_legend_param = list(title = "% of\ncluster"))
  png(paste0(out_path, "singleR_heatmap_", obj_name, ".png"), width = 12*300, height = 8*300, res = 300, type = "cairo")
  draw(ht); dev.off()
}

make_umap <- function(obj, obj_name, colors, out_path, pt_size = 1.7, label_size = 5.6) {
  ordered_levels <- natural_cluster_order(unique(as.character(obj$cell_type_label_updated)))
  obj$plot_label <- factor(obj$cell_type_label_updated, levels = ordered_levels)
  umap_reduction <- grep("umap", Reductions(obj), ignore.case = TRUE, value = TRUE)[1]
  umap_mat <- Embeddings(obj, umap_reduction)
  umap_coords <- data.frame(x = umap_mat[, 1], y = umap_mat[, 2], plot_label = obj$plot_label)
  label_centroids <- umap_coords %>% group_by(plot_label) %>% summarise(x = median(x), y = median(y), .groups = "drop")
  p <- ggplot(umap_coords, aes(x = x, y = y, color = plot_label)) +
    geom_point(size = pt_size, alpha = 0.85) +
    scale_color_manual(values = colors, name = "Cluster") +
    geom_text_repel(data = label_centroids, aes(x = x, y = y, label = plot_label),
                     inherit.aes = FALSE, size = label_size, fontface = "bold",
                     color = "black", bg.color = "white", bg.r = 0.15, max.overlaps = Inf) +
    xlab("UMAP_1") + ylab("UMAP_2") +
    ggtitle(paste0("UMAP: ", obj_name)) +
    theme_classic(base_size = 13) +
    theme(plot.title = element_text(face = "bold", size = 13), legend.position = "right",
          legend.text = element_text(size = 12, face = "bold"), legend.title = element_text(size = 12, face = "bold")) +
    guides(color = guide_legend(override.aes = list(size = 6)))
  ggsave(paste0(out_path, "UMAP_", obj_name, ".png"), p, width = 12, height = 8, dpi = 300)
}

# Seurat does not expose a fontface argument for SpatialDimPlot cluster labels, so set it on the label layers
bold_label_layers <- function(p) {
  for (i in seq_along(p$layers)) {
    if (class(p$layers[[i]]$geom)[1] %in% c("GeomTextRepel", "GeomLabelRepel", "GeomText", "GeomLabel"))
      p$layers[[i]]$aes_params$fontface <- "bold"
  }
  p
}

make_spatial <- function(obj, obj_name, colors, out_path, pt_size = 7, square = FALSE) {
  ordered_levels <- natural_cluster_order(unique(as.character(obj$cell_type_label_updated)))
  obj$plot_label <- factor(obj$cell_type_label_updated, levels = ordered_levels)
  p <- SpatialDimPlot(obj, group.by = "plot_label", cols = colors, pt.size.factor = pt_size, label = TRUE, label.size = 4.2, repel = TRUE) +
    ggtitle(paste0("Spatial clusters: ", obj_name)) +
    theme(plot.title = element_text(face = "bold", size = 13), legend.position = "right",
          legend.text = element_text(size = 12, face = "bold")) +
    guides(fill = guide_legend(override.aes = list(size = 5)))
  p <- bold_label_layers(p)
  if (square) p <- p + theme(aspect.ratio = 1)
  ggsave(paste0(out_path, "SpatialDimPlot_", obj_name, ".png"), p, width = 10, height = 8, dpi = 300)
}

# ---- Style helpers for CellChat plots (theme-only, so no argument-name risk) ----
style_cc <- function(gg, y_size = NULL) {
  th <- if (is.null(y_size)) theme(axis.text = element_text(face = "bold")) else
    theme(axis.text.x = element_text(face = "bold"), axis.text.y = element_text(size = y_size, face = "bold"))
  tryCatch(gg & th, error = function(e) gg)
}
style_geneexpr <- function(gg) tryCatch(gg & theme(axis.text.x = element_text(face = "bold"), axis.text.y = element_text(face = "bold"),
                                                   axis.title.y = element_text(size = fs(8), face = "bold"),
                                                   strip.text = element_text(size = fs(8), face = "bold")), error = function(e) gg)
bold_heatmap_names <- function(ht, row_size) {
  tryCatch({
    ht@row_names_param$gp    <- gpar(fontsize = row_size, fontface = "bold")
    ht@column_names_param$gp <- gpar(fontsize = 10, fontface = "bold")
    ht
  }, error = function(e) ht)
}

# 6b: Centrality / signaling-role heatmaps (right-side "Strength" bar removed: not comparable across conditions)
make_centrality_heatmap <- function(cc1, cc2, name1, name2, pattern, out_path) {
  pathway.union <- union(cc1@netP$pathways, cc2@netP$pathways)
  raw1 <- build_raw_centr_matrix(cc1, pathway.union, pattern)
  raw2 <- build_raw_centr_matrix(cc2, pathway.union, pattern)
  shared_top   <- max(max(colSums(raw1), colSums(raw2), na.rm = TRUE), 1)
  shared_right <- max(max(rowSums(raw1), rowSums(raw2), na.rm = TRUE), 1)

  body_h_cm <- max(8, 0.55 * length(pathway.union))     # taller body so larger pathway labels don't overlap
  fig_h     <- body_h_cm / 2.54 + 3.5
  ht1 <- netAnalysis_signalingRole_heatmap(cc1, pattern = pattern, signaling = pathway.union, title = name1, width = 5, height = body_h_cm,
                                            color.heatmap = "OrRd", color.use = identity_colors_master[levels(cc1@idents)],
                                            font.size = fs(8), ylim.top = c(0, shared_top), ylim.right = c(0, shared_right))
  ht2 <- netAnalysis_signalingRole_heatmap(cc2, pattern = pattern, signaling = pathway.union, title = name2, width = 5, height = body_h_cm,
                                            color.heatmap = "OrRd", color.use = identity_colors_master[levels(cc2@idents)],
                                            font.size = fs(8), ylim.top = c(0, shared_top), ylim.right = c(0, shared_right))
  ht1@right_annotation <- NULL; ht2@right_annotation <- NULL
  ht1 <- bold_heatmap_names(ht1, fs(8)); ht2 <- bold_heatmap_names(ht2, fs(8))

  fname_png <- paste0(out_path, pattern, "_signaling_role_heatmap_", name1, "_vs_", name2, ".png")
  fname_pdf <- paste0(out_path, pattern, "_signaling_role_heatmap_", name1, "_vs_", name2, ".pdf")
  png(fname_png, width = 13*300, height = fig_h*300, res = 300, type = "cairo"); draw(ht1 + ht2, ht_gap = unit(2, "cm")); dev.off()
  pdf(fname_pdf, width = 13, height = fig_h); draw(ht1 + ht2, ht_gap = unit(2, "cm")); dev.off()

  pathway_table <- data.frame(pathway = pathway.union, raw_sum_1 = round(rowSums(raw1), 3), raw_sum_2 = round(rowSums(raw2), 3))
  names(pathway_table)[2:3] <- c(paste0("true_sum_", name1), paste0("true_sum_", name2))
  cat("Saved:", fname_png, "\n"); print(pathway_table); cat("\n")
}

run_yap_comparison <- function(combined, conditions, out_dir) {
  dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)
  sub_obj <- subset(combined, subset = dataset %in% conditions)
  sub_obj$condition <- factor(sub_obj$dataset, levels = conditions)
  meta_df <- sub_obj@meta.data %>% rownames_to_column("cell")

  plot_list <- list()
  for (sc in names(score_labels)) if (sc %in% colnames(meta_df))
    plot_list[[sc]] <- ggplot(meta_df, aes(x=condition, y=.data[[sc]], fill=condition)) +
      geom_violin(trim=FALSE, alpha=0.6) + geom_boxplot(width=0.15, outlier.shape=NA, alpha=0.8) +
      geom_jitter(width=0.08, size=0.35, alpha=0.25) + stat_compare_means(method="wilcox.test") +
      theme_classic(base_size=13) + bold_axes(12) + labs(x=NULL, y=score_labels[[sc]], title=score_labels[[sc]])
  ggsave(paste0(out_dir,"module_scores_violin.png"), wrap_plots(plot_list), width=12, height=8, dpi=300)

  test_results <- bind_rows(lapply(names(score_labels), function(sc) if (sc %in% colnames(meta_df)) {
    wt <- wilcox.test(meta_df[[sc]] ~ meta_df$condition); data.frame(score=score_labels[[sc]], W=unname(wt$statistic), p_value=wt$p.value)
  }))
  write.csv(test_results, paste0(out_dir,"wilcoxon_test_results.csv"), row.names=FALSE)

  selected_genes <- intersect(c("Yap1","Wwtr1","Ctgf","Cyr61","Ankrd1","Tgfb1","Smad3","Serpine1","Spp1","Postn","Ptk2","Wnt5a","Axin2"), rownames(sub_obj))
  gene_theme <- theme(axis.text.x = element_text(size = fs(11), face = "bold"), axis.text.y = element_text(size = 12, face = "bold"))   # F8 gene labels ~11
  ggsave(paste0(out_dir,"key_genes_dotplot_condition.png"), DotPlot(sub_obj, features=selected_genes, group.by="condition") + RotatedAxis() + gene_theme, width=12, height=6, dpi=300)
  sub_obj$identity_condition <- paste(sub_obj$cell_identity, sub_obj$condition, sep="_")
  ggsave(paste0(out_dir,"key_genes_dotplot_identity_condition.png"), DotPlot(sub_obj, features=selected_genes, group.by="identity_condition") + RotatedAxis() + gene_theme, width=18, height=8, dpi=300)

  expr_df <- FetchData(sub_obj, vars=c(selected_genes,"identity_condition"), layer="data")
  avg_expr_mat <- t(as.matrix(expr_df %>% group_by(identity_condition) %>% summarise(across(all_of(selected_genes), ~mean(.x,na.rm=TRUE))) %>% tibble::column_to_rownames("identity_condition")))
  gene_sd <- apply(avg_expr_mat, 1, sd, na.rm=TRUE); avg_expr_mat <- avg_expr_mat[!is.na(gene_sd) & gene_sd>0, , drop=FALSE]
  rownames(avg_expr_mat) <- cap_gene(rownames(avg_expr_mat))
  mat_scaled <- t(scale(t(avg_expr_mat)))
  col_fun <- colorRamp2(c(min(mat_scaled,na.rm=TRUE),0,max(mat_scaled,na.rm=TRUE)), c("#2166AC","white","#B2182B"))
  ht <- Heatmap(mat_scaled, name="z-score", col=col_fun, column_names_rot=90,
                row_names_gp=gpar(fontsize=fs(9), fontface="bold"), column_names_gp=gpar(fontsize=9, fontface="bold"))
  png(paste0(out_dir,"average_expression_heatmap.png"), width=max(12,ncol(mat_scaled)*0.8)*300, height=9*300, res=300, type="cairo"); draw(ht); dev.off()

  cor_summary <- meta_df %>% group_by(condition) %>% summarise(
    cor_tgfb=cor(YAP_score,TGFb_score,method="spearman",use="complete.obs"),
    cor_spp1=cor(YAP_score,SPP1_score,method="spearman",use="complete.obs"),
    cor_wnt=cor(YAP_score,WNT_score,method="spearman",use="complete.obs"))
  write.csv(cor_summary, paste0(out_dir,"correlation_summary.csv"), row.names=FALSE)

  shared_identities <- intersect(unique(meta_df$cell_identity[meta_df$condition==conditions[1]]), unique(meta_df$cell_identity[meta_df$condition==conditions[2]]))
  meta_shared <- meta_df %>% filter(cell_identity %in% shared_identities) %>% group_by(cell_identity, condition) %>% filter(n()>=20) %>% ungroup()
  if (nrow(meta_shared) > 0) {
    meta_long <- meta_shared %>% select(cell, condition, cell_identity, all_of(names(score_labels))) %>%
      pivot_longer(cols=all_of(names(score_labels)), names_to="pathway", values_to="score")
    meta_long$pathway <- recode(meta_long$pathway, !!!score_labels)
    p <- ggplot(meta_long, aes(x=condition, y=score, fill=condition)) + geom_violin(trim=FALSE, alpha=0.7) +
      geom_boxplot(width=0.15, outlier.shape=NA, alpha=0.9) + stat_compare_means(method="wilcox.test", size=2.5) +
      facet_grid(pathway ~ cell_identity, scales="free_y") + theme_classic(base_size=12) +
      theme(strip.text=element_text(size=10, face="bold"), axis.text.x=element_text(angle=45,hjust=1,face="bold"), legend.position="top")
    ggsave(paste0(out_dir,"pathways_shared_identities_violin.png"), p, width=16, height=10, dpi=300)
  }
  write.csv(meta_df[,intersect(c("condition","dataset","cell_identity","cell_type_label_updated",names(score_labels)),colnames(meta_df))],
            paste0(out_dir,"module_scores_metadata.csv"), row.names=FALSE)
}

## STAGE 8: Condition-level DE -- implant-restricted AND whole-object
## (ranked by |avg_log2FC|; top-DEG dotplots split Up / Down).
## GO tables hold only terms with p.adjust < 0.05 (no qvalue column).

go_cutoff <- 0.05
run_go_bp <- function(sig_genes, comparison_name, out_dir) {
  if (length(sig_genes) < 5) return(invisible(NULL))
  ego <- tryCatch(enrichGO(gene=sig_genes, OrgDb=org.Mm.eg.db, keyType="SYMBOL", ont="BP", pAdjustMethod="BH", pvalueCutoff=0.05, qvalueCutoff=0.2), error=function(e) NULL)
  if (is.null(ego)) return(invisible(NULL))
  go_res <- ego@result[ego@result$p.adjust < go_cutoff, ]
  go_res <- go_res[, setdiff(colnames(go_res), "qvalue"), drop = FALSE]
  if (nrow(go_res) == 0) return(invisible(NULL))
  write.csv(go_res, paste0(out_dir,"GO_BP_",comparison_name,".csv"), row.names=FALSE)
  go_df <- go_res %>% arrange(p.adjust) %>% head(15)
  go_df$GeneRatioNum <- sapply(go_df$GeneRatio, function(x) eval(parse(text = x)))
  go_df$Description <- factor(go_df$Description, levels = rev(go_df$Description))
  p_go <- ggplot(go_df, aes(x = GeneRatioNum, y = Description, size = Count, color = p.adjust)) +
    geom_point() +
    scale_color_gradient(low = "red", high = "blue", name = "p.adjust") +
    scale_x_continuous(expand = expansion(mult = 0.3)) +
    theme_bw(base_size = 11) +
    labs(title = paste0("GO BP: ", comparison_name), x = "GeneRatio", y = NULL)
  save_plot_to(p_go, paste0("GO_BP_dotplot_", comparison_name), out_dir, out_dir, width=9, height=8)
}

run_de_block <- function(obj, ident1, ident2, comparison_name, out_dir, min_pct=0.1, logfc_threshold=0.25) {
  dir.create(out_dir, recursive=TRUE, showWarnings=FALSE)
  markers <- tryCatch(FindMarkers(obj, group.by="dataset", ident.1=ident1, ident.2=ident2, min.pct=min_pct, logfc.threshold=logfc_threshold, assay="RNA"), error=function(e) NULL)
  if (is.null(markers) || nrow(markers)==0) return(invisible(NULL))
  markers <- markers %>% tibble::rownames_to_column("gene") %>% arrange(desc(abs(avg_log2FC)))
  markers$sig <- "NS"; markers$sig[markers$p_val_adj<0.05 & markers$avg_log2FC>0.25] <- "Up"; markers$sig[markers$p_val_adj<0.05 & markers$avg_log2FC< -0.25] <- "Down"
  write.csv(markers, paste0(out_dir,"DEG_",comparison_name,".csv"), row.names=FALSE)   # full table; 'sig' column marks Up/Down/NS

  top_labels <- markers %>% filter(sig!="NS") %>% arrange(desc(abs(avg_log2FC))) %>% head(20)
  p_volcano <- ggplot(markers, aes(x=avg_log2FC, y=-log10(p_val_adj+1e-300), color=sig)) + geom_point(alpha=0.6, size=1) +
    scale_color_manual(values=c(Up="red",Down="blue",NS="grey70")) +
    geom_text_repel(data=top_labels, aes(label=cap_gene(gene)), size=fs(3), fontface="bold", max.overlaps=30, color="black") +
    theme_classic(base_size=13) + bold_axes(12) + labs(title=paste0("Volcano: ",comparison_name), x="log2FC", y="-log10(adj p-value)")
  save_plot_to(p_volcano, paste0("volcano_", comparison_name), out_dir, out_dir, width=9, height=8)

  top_genes_up   <- markers %>% filter(sig=="Up")   %>% arrange(desc(avg_log2FC)) %>% head(20) %>% pull(gene)
  top_genes_down <- markers %>% filter(sig=="Down") %>% arrange(avg_log2FC)       %>% head(20) %>% pull(gene)
  dot_theme <- theme(axis.text.x = element_text(size = fs(11), face = "bold"), axis.text.y = element_text(size = 12, face = "bold"))
  if (length(top_genes_up) > 0) {
    p_dot_up <- DotPlot(obj, features=top_genes_up, group.by="dataset") + RotatedAxis() +
      scale_color_gradient2(low="blue", mid="white", high="red", midpoint=0, name="Average\nExpression") +
      ggtitle(paste0("Top upregulated DEGs (by log2FC): ", comparison_name)) + dot_theme
    save_plot_to(p_dot_up, paste0("top_DEG_dotplot_UP_", comparison_name), out_dir, out_dir, width=max(12, 0.55*length(top_genes_up)+4), height=7)
  }
  if (length(top_genes_down) > 0) {
    p_dot_down <- DotPlot(obj, features=top_genes_down, group.by="dataset") + RotatedAxis() +
      scale_color_gradient2(low="blue", mid="white", high="red", midpoint=0, name="Average\nExpression") +
      ggtitle(paste0("Top downregulated DEGs (by log2FC): ", comparison_name)) + dot_theme
    save_plot_to(p_dot_down, paste0("top_DEG_dotplot_DOWN_", comparison_name), out_dir, out_dir, width=max(12, 0.55*length(top_genes_down)+4), height=7)
  }
  run_go_bp(markers %>% filter(sig!="NS") %>% pull(gene), comparison_name, out_dir)
}

run_condition_comparison <- function(combined, ident1, ident2, comparison_name, base_out_dir) {
  sub_obj <- subset(combined, subset = dataset %in% c(ident1, ident2))
  sub_obj <- subset(sub_obj, subset = implant_region == "Implant")
  run_de_block(sub_obj, ident1, ident2, paste0(comparison_name,"_whole_tissue"), paste0(base_out_dir,"whole_tissue/"))
  shared_identities <- intersect(unique(sub_obj$cell_identity[sub_obj$dataset==ident1]), unique(sub_obj$cell_identity[sub_obj$dataset==ident2]))
  for (id in shared_identities) {
    id_safe <- gsub("[^A-Za-z0-9]+", "_", id)
    sub_ct <- subset(sub_obj, subset = cell_identity == id)
    tab <- table(sub_ct$dataset)
    if (length(tab) < 2 || any(tab < 20)) next
    run_de_block(sub_ct, ident1, ident2, paste0(comparison_name,"_",id_safe), paste0(base_out_dir,"by_celltype/",id_safe,"/"))
  }
}
run_condition_comparison_wholeobject <- function(combined, ident1, ident2, comparison_name, base_out_dir) {
  sub_obj <- subset(combined, subset = dataset %in% c(ident1, ident2))
  run_de_block(sub_obj, ident1, ident2, paste0(comparison_name, "_whole_tissue"), paste0(base_out_dir, "whole_tissue/"))
  shared_identities <- intersect(unique(sub_obj$cell_identity[sub_obj$dataset == ident1]), unique(sub_obj$cell_identity[sub_obj$dataset == ident2]))
  cat("Shared identities for", comparison_name, ":\n"); print(shared_identities)
  for (id in shared_identities) {
    id_safe <- gsub("[^A-Za-z0-9]+", "_", id)
    sub_ct <- subset(sub_obj, subset = cell_identity == id)
    tab <- table(sub_ct$dataset)
    if (length(tab) < 2 || any(tab < 20)) { message("Skipping ", id, " -- insufficient cells on one side"); next }
    run_de_block(sub_ct, ident1, ident2, paste0(comparison_name, "_", id_safe), paste0(base_out_dir, "by_celltype/", id_safe, "/"))
  }
}

## STAGE 9: IMM-1 / FIB-1 trackplots (within-pool DE; p_val_adj <= 0.05, top 10 by pct.1)
## Gene names 27 pt bold-italic (2x the 13.5 pt F9 size), cluster labels 28 pt bold
################################################################################
make_trackplot <- function(cluster_info, out_filename, plot_title, custom_colors = NULL) {
  cluster_order <- sapply(cluster_info, function(x) x$plot_label)
  cluster_colors <- if (!is.null(custom_colors)) custom_colors[cluster_order] else
    setNames(celltype_colors_master[sapply(cluster_info, function(x) x$cluster)], cluster_order)

  subset_list <- list()
  for (cl in cluster_info) { s <- subset(cl$obj, subset = cell_type_label_updated == cl$cluster); s$plot_label <- cl$plot_label; subset_list[[cl$plot_label]] <- s }
  pool <- merge(subset_list[[1]], y = subset_list[-1])
  DefaultAssay(pool) <- "RNA"; pool <- JoinLayers(pool, assay="RNA")
  if (nrow(GetAssayData(pool, assay="RNA", layer="data"))==0) pool <- NormalizeData(pool, assay="RNA")
  Idents(pool) <- factor(pool$plot_label, levels = cluster_order)

  pool_markers <- FindAllMarkers(pool, only.pos=TRUE, min.pct=0.25, logfc.threshold=0.25)
  combined_top10_annot <- pool_markers %>% filter(p_val_adj <= 0.05) %>% group_by(cluster) %>%
    arrange(desc(pct.1), .by_group=TRUE) %>% slice_head(n=10) %>% ungroup()
  gene_order <- combined_top10_annot %>% arrange(cluster, desc(pct.1)) %>% pull(gene) %>% unique()
  gene_cluster_map <- combined_top10_annot %>% dplyr::select(gene, cluster) %>% distinct(gene, .keep_all=TRUE)

  n_bins <- 10; bin_data_list <- list()
  for (cl in cluster_info) {
    DefaultAssay(cl$obj) <- "RNA"
    cells <- colnames(cl$obj)[cl$obj@meta.data[["cell_type_label_updated"]]==cl$cluster]
    sub_obj <- subset(cl$obj, cells=cells)
    genes_present <- gene_order[gene_order %in% rownames(sub_obj)]; if (length(genes_present)==0) next
    expr_mat <- as.matrix(GetAssayData(sub_obj, assay="RNA", layer="data")[genes_present, , drop=FALSE])
    n_cells <- ncol(expr_mat); bin_size <- ceiling(n_cells/n_bins); bin_df_list <- list()
    for (g in genes_present) { sorted_expr <- sort(expr_mat[g,])
      for (b in 1:n_bins) { start_idx <- (b-1)*bin_size+1; end_idx <- min(b*bin_size, n_cells); if (start_idx>n_cells) break
        bin_df_list[[paste(g,b)]] <- data.frame(gene=g, bin=b, bin_expr=mean(sorted_expr[start_idx:end_idx]), cluster_label=cl$plot_label, stringsAsFactors=FALSE) } }
    if (length(bin_df_list)>0) bin_data_list[[cl$plot_label]] <- bind_rows(bin_df_list)
  }
  bin_df_all <- bind_rows(bin_data_list) %>% group_by(gene) %>% mutate(expr_scaled=(bin_expr-min(bin_expr))/(max(bin_expr)-min(bin_expr)+1e-10)) %>% ungroup() %>%
    left_join(gene_cluster_map, by="gene")
  bin_df_all$fill_color <- cluster_colors[bin_df_all$cluster]
  bin_df_all$gene <- factor(as.character(bin_df_all$gene), levels=rev(gene_order))
  bin_df_all$cluster_label <- factor(bin_df_all$cluster_label, levels=cluster_order)
  bin_df_all <- bin_df_all %>% mutate(gene_num=as.numeric(gene), cluster_num=as.numeric(cluster_label),
    x_pos=cluster_num-0.4+(bin-1)/(n_bins-1)*0.8, y_base=gene_num, y_top=gene_num+expr_scaled*0.85)

  n_total_genes <- length(gene_order)
  gene_groups <- combined_top10_annot %>% arrange(cluster, desc(pct.1)) %>% distinct(gene, .keep_all=TRUE) %>%
    group_by(cluster) %>% summarise(n_genes=n(), .groups="drop") %>%
    mutate(end_pos=cumsum(n_genes), start_pos=end_pos-n_genes+1, y_start=n_total_genes-end_pos+0.5, y_end=n_total_genes-start_pos+1.5,
           y_mid=(y_start+y_end)/2, fill_color=cluster_colors[cluster], short_label=cap_label(as.character(cluster)))
  y_colors_final <- cluster_colors[gene_cluster_map$cluster[match(levels(bin_df_all$gene), gene_cluster_map$gene)]]
  x_ann_start <- length(cluster_order)+0.6; x_ann_end <- length(cluster_order)+1.2

  gene_size <- fs(9) * 2          # 27 pt
  xlab_size <- 14 * 2             # 28 pt
  p <- ggplot(bin_df_all) +
    geom_ribbon(aes(x=x_pos, ymin=y_base, ymax=y_top, fill=fill_color, group=interaction(gene,cluster_label)), alpha=0.85) +
    scale_fill_identity() +
    geom_segment(data=bin_df_all %>% distinct(gene,gene_num,cluster_num) %>% mutate(x_start=cluster_num-0.45, x_end=cluster_num+0.45),
                 aes(x=x_start, xend=x_end, y=gene_num, yend=gene_num), color="grey70", linewidth=0.3) +
    geom_rect(data=gene_groups, aes(xmin=x_ann_start, xmax=x_ann_end, ymin=y_start, ymax=y_end, fill=fill_color), alpha=0.85, inherit.aes=FALSE) +
    geom_text(data=gene_groups, aes(x=(x_ann_start+x_ann_end)/2, y=y_mid, label=short_label), size=4.5, fontface="bold", color="white", inherit.aes=FALSE) +
    geom_hline(data=gene_groups %>% filter(y_start>0.5), aes(yintercept=y_start-0.5), color="white", linewidth=0.8, inherit.aes=FALSE) +
    scale_x_continuous(breaks=1:length(cluster_order), labels=cap_label(cluster_order), expand=c(0.02,0.02), limits=c(0.5,x_ann_end+0.1)) +
    scale_y_continuous(breaks=1:n_total_genes, labels=cap_gene(levels(bin_df_all$gene)), expand=c(0.01,0.5)) +
    theme_bw(base_size=12) +
    theme(axis.text.x=element_text(angle=45,hjust=1,size=xlab_size,face="bold"),
          axis.text.y=element_text(size=gene_size,face="bold.italic",color=y_colors_final),
          axis.title.x=element_blank(), axis.title.y=element_text(size=18,face="bold"),
          plot.title=element_text(size=24,face="bold"), plot.subtitle=element_text(size=16),
          panel.grid=element_blank(), legend.position="none") +
    ylab("Average gene expression per cluster") + ggtitle(plot_title, subtitle="Right panel = source cluster")

  png(paste0(trackplot_dir9, out_filename), width=3400, height=max(6000, n_total_genes*210+1800), res=150); print(p); dev.off()
}

make_composition_plot <- function(combined, conditions, out_dir, plot_name) {
  sub_df <- combined@meta.data %>% filter(dataset %in% conditions) %>% mutate(dataset = factor(dataset, levels = conditions))
  ids_by_condition <- lapply(conditions, function(cond) unique(sub_df$cell_identity[sub_df$dataset == cond]))
  shared_ids <- Reduce(intersect, ids_by_condition)
  unique_ids <- setdiff(unique(sub_df$cell_identity), shared_ids)
  identity_levels <- c(intersect(master_identity_order, shared_ids), intersect(master_identity_order, unique_ids))
  sub_df$cell_identity <- factor(sub_df$cell_identity, levels = identity_levels)

  p <- ggplot(sub_df, aes(x = dataset, fill = cell_identity)) +
    geom_bar(position = "fill", width = 0.7, color = "white", linewidth = 0.2) +
    scale_y_continuous(labels = percent_format(), expand = c(0, 0)) +
    scale_fill_manual(values = identity_colors_master[identity_levels], name = "Cell identity") +
    theme_classic(base_size = 13) +
    theme(axis.title.x = element_blank(), axis.text.x = element_text(size = 12, face = "bold"), axis.text.y = element_text(face = "bold"),
          legend.key.size = unit(0.45, "cm"), legend.text = element_text(size = 11, face = "bold"), legend.title = element_text(face = "bold")) +

compute_top_markers_per_object <- function(obj, n_top = 2) {
  obj_de <- obj
  DefaultAssay(obj_de) <- "RNA"
  if (nrow(GetAssayData(obj_de, assay = "RNA", layer = "data")) == 0) obj_de <- NormalizeData(obj_de, assay = "RNA")
  Idents(obj_de) <- obj_de$cell_type_label_updated
  markers <- FindAllMarkers(obj_de, only.pos = TRUE, min.pct = 0.25, logfc.threshold = 0.25)
  markers %>% filter(p_val_adj <= 0.05) %>% group_by(cluster) %>% arrange(desc(avg_log2FC), .by_group = TRUE) %>% slice_head(n = n_top) %>% ungroup()
}

make_pair_trackplot_v2 <- function(obj1, obj1_name, obj2, obj2_name, top_markers_by_obj, out_filename, plot_title) {
  clusters1 <- natural_cluster_order(unique(as.character(obj1$cell_type_label_updated)))
  clusters2 <- natural_cluster_order(unique(as.character(obj2$cell_type_label_updated)))
  clusters_df <- bind_rows(
    data.frame(obj_name = obj1_name, cluster = clusters1, identity = unname(label_to_identity[clusters1]), stringsAsFactors = FALSE),
    data.frame(obj_name = obj2_name, cluster = clusters2, identity = unname(label_to_identity[clusters2]), stringsAsFactors = FALSE))
  clusters_df$identity <- factor(clusters_df$identity, levels = intersect(master_identity_order, unique(clusters_df$identity)))
  clusters_df$obj_name <- factor(clusters_df$obj_name, levels = c(obj1_name, obj2_name))
  clusters_df <- clusters_df %>% arrange(identity, obj_name)
  clusters_df$cluster_num <- seq_len(nrow(clusters_df))
  clusters_df$stripe_color <- dataset_colors[as.character(clusters_df$obj_name)]

  obj_lookup <- setNames(list(obj1, obj2), c(obj1_name, obj2_name))
  cluster_info <- lapply(seq_len(nrow(clusters_df)), function(i) {
    list(obj = obj_lookup[[as.character(clusters_df$obj_name[i])]], cluster = clusters_df$cluster[i],
         plot_label = paste(clusters_df$obj_name[i], clusters_df$cluster[i]))
  })
  cluster_order <- sapply(cluster_info, function(x) x$plot_label)
  raw_cluster_lookup <- setNames(clusters_df$cluster, cluster_order)
  cluster_colors <- celltype_colors_master[raw_cluster_lookup]
  names(cluster_colors) <- cluster_order

  top_markers_annot <- bind_rows(
    top_markers_by_obj[[obj1_name]] %>% filter(cluster %in% clusters1) %>% mutate(plot_label = paste(obj1_name, cluster)),
    top_markers_by_obj[[obj2_name]] %>% filter(cluster %in% clusters2) %>% mutate(plot_label = paste(obj2_name, cluster)))
  top_markers_annot$plot_label <- factor(top_markers_annot$plot_label, levels = cluster_order)
  top_markers_annot <- top_markers_annot %>% arrange(plot_label, desc(avg_log2FC))

  gene_position_df <- top_markers_annot %>% distinct(gene, .keep_all = TRUE) %>%
    transmute(gene, owning_cluster = plot_label) %>% mutate(row_num = row_number())
  gene_order <- gene_position_df$gene
  n_total_genes <- nrow(gene_position_df)
  gene_row_lookup     <- setNames(gene_position_df$row_num, gene_position_df$gene)
  gene_cluster_lookup <- setNames(as.character(gene_position_df$owning_cluster), gene_position_df$gene)
  gene_labels_ordered <- rev(gene_position_df$gene)

  n_bins <- 10; bin_data_list <- list()
  for (cl in cluster_info) {
    DefaultAssay(cl$obj) <- "RNA"
    cells <- colnames(cl$obj)[cl$obj@meta.data[["cell_type_label_updated"]] == cl$cluster]
    sub_obj <- subset(cl$obj, cells = cells)
    genes_present <- gene_order[gene_order %in% rownames(sub_obj)]; if (length(genes_present) == 0) next
    expr_mat <- as.matrix(GetAssayData(sub_obj, assay = "RNA", layer = "data")[genes_present, , drop = FALSE])
    n_cells <- ncol(expr_mat); bin_size <- ceiling(n_cells / n_bins); bin_df_list <- list()
    for (g in genes_present) { sorted_expr <- sort(expr_mat[g, ])
      for (b in 1:n_bins) { start_idx <- (b - 1) * bin_size + 1; end_idx <- min(b * bin_size, n_cells); if (start_idx > n_cells) break

    if (length(bin_df_list) > 0) bin_data_list[[cl$plot_label]] <- bind_rows(bin_df_list)
  }
  bin_df_all <- bind_rows(bin_data_list) %>% group_by(gene) %>% mutate(expr_scaled = (bin_expr - min(bin_expr)) / (max(bin_expr) - min(bin_expr) + 1e-10)) %>% ungroup()
  bin_df_all$fill_color <- cluster_colors[gene_cluster_lookup[bin_df_all$gene]]

  y_colors_final <- cluster_colors[gene_cluster_lookup[gene_labels_ordered]]
  x_ann_start <- length(cluster_order) + 0.6; x_ann_end <- length(cluster_order) + 1.2

  gene_size <- fs(9) * 2          # 27 pt
  xlab_size <- 12 * 2             # 24 pt
  p <- ggplot(bin_df_all) +
    geom_rect(data = clusters_df, aes(xmin = cluster_num - 0.48, xmax = cluster_num + 0.48, ymin = 0.5, ymax = n_total_genes + 0.5, fill = stripe_color), alpha = 0.22, inherit.aes = FALSE) +
    geom_ribbon(aes(x = x_pos, ymin = y_base, ymax = y_top, fill = fill_color, group = interaction(gene, cluster_label)), alpha = 0.9) +
    scale_fill_identity() +
    geom_segment(data = bin_df_all %>% distinct(gene, gene_num, cluster_num) %>% mutate(x_start = cluster_num - 0.45, x_end = cluster_num + 0.45),
                 aes(x = x_start, xend = x_end, y = gene_num, yend = gene_num), color = "grey70", linewidth = 0.3) +
    geom_rect(data = gene_groups, aes(xmin = x_ann_start, xmax = x_ann_end, ymin = y_start, ymax = y_end, fill = fill_color), alpha = 0.85, inherit.aes = FALSE) +
    geom_text(data = gene_groups, aes(x = (x_ann_start + x_ann_end) / 2, y = y_mid, label = short_label), size = 2.6, fontface = "bold", color = "white", inherit.aes = FALSE, angle = 60) +
    geom_hline(data = gene_groups %>% filter(y_start > 0.5), aes(yintercept = y_start + 0.5), color = "white", linewidth = 0.8, inherit.aes = FALSE) +
    annotate("text", x = 1, y = n_total_genes + 1.8, label = paste0("\u25A0 ", cap_label(obj1_name)), color = dataset_colors[obj1_name], fontface = "bold", size = 6, hjust = 0) +
    annotate("text", x = 1, y = n_total_genes + 1.2, label = paste0("\u25A0 ", cap_label(obj2_name)), color = dataset_colors[obj2_name], fontface = "bold", size = 6, hjust = 0) +
    scale_x_continuous(breaks = 1:length(cluster_order), labels = cap_label(cluster_order), expand = c(0.02, 0.02), limits = c(0.5, x_ann_end + 0.1)) +
    scale_y_continuous(breaks = 1:n_total_genes, labels = cap_gene(gene_labels_ordered), expand = expansion(mult = c(0.01, 0.08))) +
    theme_bw(base_size = 12) +
    theme(axis.text.x = element_text(angle = 45, hjust = 1, size = xlab_size, face = "bold"),
          axis.text.y = element_text(size = gene_size, face = "bold.italic", color = y_colors_final),
          axis.title.x = element_blank(), axis.title.y = element_text(size = 18, face = "bold"),
          plot.title = element_text(size = 24, face = "bold"), plot.subtitle = element_text(size = 16),
          panel.grid = element_blank(), legend.position = "none") +
    ylab("Average gene expression per cluster") + ggtitle(plot_title, subtitle = "Background stripe color = source object (see key, top-left)")

  out_width  <- max(3600, length(cluster_order) * 135 + 900)
  out_height <- max(4000, n_total_genes * 110 + 1500)
  png(paste0(trackplot_dir9, out_filename), width = out_width, height = out_height, res = 150); print(p); dev.off()
}

get_pathway_genes <- function(pathway_name, db = CellChatDB.mouse) {
  all_pathways <- unique(db$interaction$pathway_name)
  match_name <- all_pathways[match(tolower(pathway_name), tolower(all_pathways))]
  if (is.na(match_name)) { warning("Pathway '", pathway_name, "' not found"); return(character(0)) }
  interaction_sub <- db$interaction[db$interaction$pathway_name == match_name, ]
  raw_genes <- unique(c(interaction_sub$ligand, interaction_sub$receptor))
  resolved <- unlist(lapply(raw_genes, function(g) {
    if (g %in% rownames(db$complex)) { subunits <- unlist(db$complex[g, ]); subunits[subunits != "" & !is.na(subunits)] } else { g }
  }))
  unique(resolved)
}
resolve_gene_symbol <- function(gene, obj) {
  rn <- rownames(obj); idx <- match(tolower(gene), tolower(rn)); if (is.na(idx)) return(NA_character_); rn[idx]
}
placeholder_plot <- function(label_text) {
  ggplot() + theme_void() + annotate("text", x = 0.5, y = 0.5, label = paste0(label_text, "\n(not detected)"), size = 4.5, fontface = "bold") + xlim(0, 1) + ylim(0, 1)
}
make_gene_violin <- function(obj, gene, title, subtitle = NULL, y_limits = NULL) {
  ordered_levels <- natural_cluster_order(unique(as.character(obj$cell_type_label_updated)))
  colors_this <- celltype_colors_master[ordered_levels]; colors_this[is.na(colors_this)] <- "grey70"; names(colors_this) <- ordered_levels
  expr <- FetchData(obj, vars = gene, layer = "data")[, 1]; cluster_vec <- as.character(obj$cell_type_label_updated)
  clusters_expressed <- ordered_levels[sapply(ordered_levels, function(cl) any(expr[cluster_vec == cl] > 0))]
  if (length(clusters_expressed) == 0) return(NULL)
  keep_cells <- cluster_vec %in% clusters_expressed
  df <- data.frame(expr = expr[keep_cells], cluster = factor(cluster_vec[keep_cells], levels = clusters_expressed))
  p <- ggplot(df, aes(x = cluster, y = expr, fill = cluster)) + geom_violin(scale = "width", trim = TRUE) +
    geom_jitter(width = 0.15, size = 0.3, alpha = 0.3) + scale_fill_manual(values = colors_this[clusters_expressed]) +
    theme_classic(base_size = 11) +
    theme(axis.text.x = element_text(angle = 45, hjust = 1, size = 10, face = "bold"), axis.text.y = element_text(face = "bold"),
          plot.title = element_text(size = fs(13), face = "bold"), plot.subtitle = element_text(size = 11), legend.position = "none") +
    labs(title = title, subtitle = subtitle, x = NULL, y = "Expression")
  if (!is.null(y_limits)) p <- p + coord_cartesian(ylim = y_limits)
  p
}

  if (square) p_spatial <- p_spatial & theme(aspect.ratio = 1)

  ggsave(paste0(marker_spatial_violin_dir9, "violin_", file_tag, ".png"), wrap_plots(plot_list, ncol = ncol_use),
         width = max(6, n_clusters_max * 0.5) * ncol_use, height = 5 * ceiling(length(plot_list) / ncol_use), dpi = 300)
}

# Part B/C: pairwise gene comparison. Spatial always uses the WHOLE objects (obj_meta);
# violins / averages use `stat_meta` (whole objects, or implant no-FIB-1 subsets when implant = TRUE).
make_gene_comparison_plots <- function(genes, obj1_name, obj2_name, note, file_prefix, out_dir, stat_meta = obj_meta, implant = FALSE) {
  info1 <- obj_meta[[obj1_name]]; info2 <- obj_meta[[obj2_name]]
  s1 <- stat_meta[[obj1_name]]$obj; s2 <- stat_meta[[obj2_name]]$obj
  resolved1 <- sapply(genes, resolve_gene_symbol, obj = info1$obj); resolved2 <- sapply(genes, resolve_gene_symbol, obj = info2$obj)
  resolved_s1 <- sapply(genes, resolve_gene_symbol, obj = s1);       resolved_s2 <- sapply(genes, resolve_gene_symbol, obj = s2)
  keep <- !is.na(resolved1) & !is.na(resolved2) & !is.na(resolved_s1) & !is.na(resolved_s2); genes_use <- genes[keep]
  if (any(!keep)) message("Genes not found, skipped: ", paste(genes[!keep], collapse = ", "))
  if (length(genes_use) == 0) { message("No usable genes for ", file_prefix); return(invisible(NULL)) }

run_cluster_de_block <- function(obj, group_col, ident1, ident2, comparison_name, out_dir, min_pct=0.1, logfc_threshold=0.25) {
  dir.create(out_dir, recursive=TRUE, showWarnings=FALSE)
  markers <- tryCatch(FindMarkers(obj, group.by=group_col, ident.1=ident1, ident.2=ident2, min.pct=min_pct, logfc.threshold=logfc_threshold, assay="RNA"), error=function(e) NULL)
  if (is.null(markers) || nrow(markers)==0) return(invisible(NULL))
  markers <- markers %>% tibble::rownames_to_column("gene") %>% arrange(p_val_adj)
  markers$sig <- "NS"; markers$sig[markers$p_val_adj<0.05 & markers$avg_log2FC>0.25] <- "Up"; markers$sig[markers$p_val_adj<0.05 & markers$avg_log2FC< -0.25] <- "Down"
  write.csv(markers, paste0(out_dir,"DEG_",comparison_name,".csv"), row.names=FALSE)
  top_labels <- markers %>% filter(sig!="NS") %>% arrange(p_val_adj) %>% head(20)
  p_volcano <- ggplot(markers, aes(x=avg_log2FC, y=-log10(p_val_adj+1e-300), color=sig)) + geom_point(alpha=0.6, size=1) +
    scale_color_manual(values=c(Up="red",Down="blue",NS="grey70")) +
    geom_text_repel(data=top_labels, aes(label=cap_gene(gene)), size=fs(3), fontface="bold", max.overlaps=30, color="black") +
    theme_classic(base_size=13) + bold_axes(12) + labs(title=paste0("Volcano: ",comparison_name), x="log2FC", y="-log10(adj p-value)")
  save_plot_to(p_volcano, paste0("volcano_", comparison_name), out_dir, out_dir, width=9, height=8)

  top_genes <- c(markers %>% filter(sig=="Up") %>% arrange(p_val_adj) %>% head(15) %>% pull(gene),
                 markers %>% filter(sig=="Down") %>% arrange(p_val_adj) %>% head(15) %>% pull(gene))
  if (length(top_genes)>0) {
    p_dot <- DotPlot(obj, features=top_genes, group.by=group_col) + RotatedAxis() +
      scale_color_gradient2(low="blue", mid="white", high="red", midpoint=0, name="Average\nExpression") + ggtitle(paste0("Top DEGs: ",comparison_name)) +
      theme(axis.text.x = element_text(size = fs(11), face = "bold"), axis.text.y = element_text(size = 12, face = "bold"))
    save_plot_to(p_dot, paste0("top_DEG_dotplot_", comparison_name), out_dir, out_dir, width=max(12, 0.55*length(top_genes)+4), height=7)
  }
  run_go_bp(markers %>% filter(sig!="NS") %>% pull(gene), comparison_name, out_dir)
}
run_intra_object_cluster_comparison <- function(obj, obj_name, cluster1, cluster2, comparison_tag, out_dir) {
  sub_obj <- subset(obj, subset = cell_type_label_updated %in% c(cluster1, cluster2))
  sub_obj$cell_type_label_updated <- droplevels(factor(sub_obj$cell_type_label_updated))
  run_cluster_de_block(sub_obj, "cell_type_label_updated", cluster1, cluster2, paste0(obj_name, "_", comparison_tag), out_dir)
}
run_all_pairwise <- function(obj, obj_name, cluster_vec, lineage_tag, out_dir) {
  if (length(cluster_vec) < 2) { message("Skipping ", lineage_tag, " for ", obj_name, " -- only ", length(cluster_vec), " cluster(s)"); return(invisible(NULL)) }
  for (pr in combn(cluster_vec, 2, simplify = FALSE)) {
    c1 <- pr[1]; c2 <- pr[2]
    comparison_tag <- paste0(lineage_tag, "_", gsub("[^A-Za-z0-9]+", "_", c1), "_vs_", gsub("[^A-Za-z0-9]+", "_", c2))
    run_intra_object_cluster_comparison(obj, obj_name, c1, c2, comparison_tag, paste0(out_dir, obj_name, "/"))
  }
}

run_module_score_comparison <- function(combined, conditions, comparison_tag, out_dir) {
  sub_obj <- subset(combined, subset = dataset %in% conditions)
  sub_obj$dataset <- factor(sub_obj$dataset, levels = conditions)
  identity_levels <- intersect(master_identity_order, unique(sub_obj$cell_identity))
  sub_obj$cell_identity <- factor(sub_obj$cell_identity, levels = identity_levels)
  plot_df <- FetchData(sub_obj, vars = c("cell_identity", "dataset", modules))
  plot_df_long <- plot_df %>%
    pivot_longer(cols = all_of(modules), names_to = "module", values_to = "score") %>%
    mutate(module_label = module_names[module], dataset = factor(dataset, levels = conditions), cell_identity = factor(cell_identity, levels = identity_levels))
  colors_use <- dataset_colors[conditions]
  axis_th <- theme(axis.text.x = element_text(angle = 45, hjust = 1, size = 12, face = "bold"), axis.text.y = element_text(size = 12, face = "bold"))

  for (mod in modules) {
    df_mod <- plot_df_long %>% filter(module == mod, !is.na(score))
    stat_df <- tryCatch({
      df_mod %>% group_by(cell_identity) %>% filter(n_distinct(dataset) == 2) %>%
        wilcox_test(score ~ dataset) %>% add_significance("p") %>% add_xy_position(x = "cell_identity", dodge = 0.85)
    }, error = function(e) data.frame())
    p <- ggplot(df_mod, aes(x = cell_identity, y = score, fill = dataset)) +
      geom_violin(position = position_dodge(width = 0.85), trim = TRUE, alpha = 0.75, scale = "width") +
      geom_boxplot(position = position_dodge(width = 0.85), width = 0.15, outlier.shape = NA, alpha = 0.9) +
      scale_fill_manual(values = colors_use) + theme_classic() + axis_th +
      theme(plot.title = element_text(size = 16, face = "bold")) +
      labs(title = paste0(module_names[mod], ": ", conditions[1], " vs ", conditions[2], " per cell type"), x = NULL, y = "Module score", fill = "Object")
    if (nrow(stat_df) > 0) p <- p + stat_pvalue_manual(stat_df, label = "p.signif", hide.ns = TRUE, tip.length = 0.01)
    ggsave(paste0(out_dir, "Violin_Boxplot_Stats_", mod, "_", comparison_tag, ".jpeg"), p, width = 11, height = 7, dpi = 300)
    ggsave(paste0(out_dir, "Violin_Boxplot_Stats_", mod, "_", comparison_tag, ".pdf"),  p, width = 11, height = 7)
  }
  p_all_stats <- ggplot(plot_df_long, aes(x = cell_identity, y = score, fill = dataset)) +
    geom_violin(position = position_dodge(width = 0.85), trim = TRUE, alpha = 0.75, scale = "width") +
    geom_boxplot(position = position_dodge(width = 0.85), width = 0.15, outlier.shape = NA, alpha = 0.9) +
    facet_wrap(~ module_label, scales = "free_y", ncol = 2) +
    scale_fill_manual(values = colors_use) + theme_classic() + axis_th +
    theme(strip.text = element_text(size = 13, face = "bold"), plot.title = element_text(size = 17, face = "bold")) +
    labs(title = paste0("Module scores across all cell identities: ", conditions[1], " vs ", conditions[2]), x = NULL, y = "Module score", fill = "Object")
  ggsave(paste0(out_dir, "Violin_Boxplot_ALL_Modules_", comparison_tag, ".jpeg"), p_all_stats, width = 15, height = 10, dpi = 300)
  ggsave(paste0(out_dir, "Violin_Boxplot_ALL_Modules_", comparison_tag, ".pdf"),  p_all_stats, width = 15, height = 10)
}

run_module_score_comparison_common <- function(combined, conditions, comparison_tag, out_dir) {
  sub_obj <- subset(combined, subset = dataset %in% conditions)
  sub_obj$dataset <- factor(sub_obj$dataset, levels = conditions)
  shared_identities <- intersect(unique(sub_obj$cell_identity[sub_obj$dataset == conditions[1]]),
                                 unique(sub_obj$cell_identity[sub_obj$dataset == conditions[2]]))
  sub_obj <- subset(sub_obj, subset = cell_identity %in% shared_identities)
  identity_levels <- intersect(master_identity_order, shared_identities)
  sub_obj$cell_identity <- factor(sub_obj$cell_identity, levels = identity_levels)
  plot_df <- FetchData(sub_obj, vars = c("cell_identity", "dataset", modules))
  plot_df_long <- plot_df %>%
    pivot_longer(cols = all_of(modules), names_to = "module", values_to = "score") %>%
    mutate(module_label = module_names[module], dataset = factor(dataset, levels = conditions), cell_identity = factor(cell_identity, levels = identity_levels))
  colors_use <- dataset_colors[conditions]
  axis_th <- theme(axis.text.x = element_text(angle = 45, hjust = 1, size = 12, face = "bold"), axis.text.y = element_text(size = 12, face = "bold"))
  for (mod in modules) {
    df_mod <- plot_df_long %>% filter(module == mod, !is.na(score))
    stat_df <- tryCatch({
      df_mod %>% group_by(cell_identity) %>% filter(n_distinct(dataset) == 2) %>%
        wilcox_test(score ~ dataset) %>% add_significance("p") %>% add_xy_position(x = "cell_identity", dodge = 0.85)
    }, error = function(e) data.frame())
    p <- ggplot(df_mod, aes(x = cell_identity, y = score, fill = dataset)) +
      geom_violin(position = position_dodge(width = 0.85), trim = TRUE, alpha = 0.75, scale = "width") +
      geom_boxplot(position = position_dodge(width = 0.85), width = 0.15, outlier.shape = NA, alpha = 0.9) +
      scale_fill_manual(values = colors_use) + theme_classic() + axis_th + theme(plot.title = element_text(size = 16, face = "bold")) +
      labs(title = paste0(module_names[mod], ": ", conditions[1], " vs ", conditions[2], " (shared cell identities only)"), x = NULL, y = "Module score", fill = "Object")
    if (nrow(stat_df) > 0) p <- p + stat_pvalue_manual(stat_df, label = "p.signif", hide.ns = TRUE, tip.length = 0.01)
    ggsave(paste0(out_dir, "Violin_Boxplot_Stats_", mod, "_", comparison_tag, "_common_clusters.jpeg"), p, width = 11, height = 7, dpi = 300)
    ggsave(paste0(out_dir, "Violin_Boxplot_Stats_", mod, "_", comparison_tag, "_common_clusters.pdf"),  p, width = 11, height = 7)
  }
  p_all <- ggplot(plot_df_long, aes(x = cell_identity, y = score, fill = dataset)) +
    geom_violin(position = position_dodge(width = 0.85), trim = TRUE, alpha = 0.75, scale = "width") +
    geom_boxplot(position = position_dodge(width = 0.85), width = 0.15, outlier.shape = NA, alpha = 0.9) +
    facet_wrap(~ module_label, scales = "free_y", ncol = 2) +
    scale_fill_manual(values = colors_use) + theme_classic() + axis_th +
    theme(strip.text = element_text(size = 13, face = "bold"), plot.title = element_text(size = 17, face = "bold")) +
    labs(title = paste0("Module scores, shared cell identities only: ", conditions[1], " vs ", conditions[2]), x = NULL, y = "Module score", fill = "Object")
  ggsave(paste0(out_dir, "Violin_Boxplot_ALL_Modules_", comparison_tag, "_common_clusters.jpeg"), p_all, width = 15, height = 10, dpi = 300)
  ggsave(paste0(out_dir, "Violin_Boxplot_ALL_Modules_", comparison_tag, "_common_clusters.pdf"),  p_all, width = 15, height = 10)
}

compute_full_markers <- function(obj) {
  obj_de <- obj
  DefaultAssay(obj_de) <- "RNA"
  if (nrow(GetAssayData(obj_de, assay = "RNA", layer = "data")) == 0) obj_de <- NormalizeData(obj_de, assay = "RNA")
  Idents(obj_de) <- obj_de$cell_type_label_updated
  FindAllMarkers(obj_de, only.pos = TRUE, min.pct = 0.25, logfc.threshold = 0.25)
}
get_sig_markers <- function(full_markers) full_markers %>% filter(p_val_adj <= 0.05) %>% arrange(cluster, desc(avg_log2FC))

    mutate(rank_in_cluster = row_number()) %>% slice_head(n = 10) %>% ungroup()
}

  obj

make_pair_trackplot_module_score <- function(obj1_scored, obj1_name, obj2_scored, obj2_name, out_filename, plot_title) {
  clusters1 <- natural_cluster_order(unique(as.character(obj1_scored$cell_type_label_updated)))
  clusters2 <- natural_cluster_order(unique(as.character(obj2_scored$cell_type_label_updated)))
  clusters_df <- bind_rows(
    data.frame(obj_name = obj1_name, cluster = clusters1, identity = unname(label_to_identity[clusters1]), stringsAsFactors = FALSE),
    data.frame(obj_name = obj2_name, cluster = clusters2, identity = unname(label_to_identity[clusters2]), stringsAsFactors = FALSE))
  clusters_df$identity <- factor(clusters_df$identity, levels = intersect(master_identity_order, unique(clusters_df$identity)))
  clusters_df$obj_name <- factor(clusters_df$obj_name, levels = c(obj1_name, obj2_name))
  clusters_df <- clusters_df %>% arrange(identity, obj_name)
  clusters_df$cluster_num  <- seq_len(nrow(clusters_df))
  clusters_df$stripe_color <- dataset_colors[as.character(clusters_df$obj_name)]
  clusters_df$plot_label   <- paste(cap_label(as.character(clusters_df$obj_name)), clusters_df$cluster)   # "Rod wk4 C1-IMM-1"
  clusters_df$module_col   <- paste0(as.character(clusters_df$obj_name), "_", clusters_df$cluster)         # internal column name
  obj_lookup     <- setNames(list(obj1_scored, obj2_scored), c(obj1_name, obj2_name))
  plot_order     <- clusters_df$plot_label
  cluster_colors <- setNames(celltype_colors_master[clusters_df$cluster], plot_order)
  n_total        <- length(plot_order)

  n_bins <- 10; bin_data_list <- list()
  for (i in seq_len(nrow(clusters_df))) {
    obj_this <- obj_lookup[[as.character(clusters_df$obj_name[i])]]
    cells    <- colnames(obj_this)[obj_this@meta.data[["cell_type_label_updated"]] == clusters_df$cluster[i]]
    meta_sub <- obj_this@meta.data[cells, , drop = FALSE]
    for (j in seq_len(nrow(clusters_df))) {
      mod_col <- clusters_df$module_col[j]
      if (!(mod_col %in% colnames(meta_sub))) next
      mod_vals <- meta_sub[[mod_col]]
      n_cells  <- length(mod_vals); if (n_cells == 0) next
      bin_size <- ceiling(n_cells / n_bins); sorted_vals <- sort(mod_vals)
      for (b in 1:n_bins) {
        start_idx <- (b - 1) * bin_size + 1; end_idx <- min(b * bin_size, n_cells); if (start_idx > n_cells) break
        bin_data_list[[paste(i, j, b)]] <- data.frame(row_label = plot_order[j], bin = b, bin_score = mean(sorted_vals[start_idx:end_idx]),
                                                      col_label = plot_order[i], stringsAsFactors = FALSE)
      }
    }
  }
  bin_df_all <- bind_rows(bin_data_list) %>% group_by(row_label) %>%
    mutate(score_scaled = (bin_score - min(bin_score)) / (max(bin_score) - min(bin_score) + 1e-10)) %>% ungroup()
  bin_df_all$fill_color  <- cluster_colors[bin_df_all$row_label]
  bin_df_all$row_num     <- match(bin_df_all$row_label, plot_order)
  bin_df_all$row_num_rev <- n_total - bin_df_all$row_num + 1
  bin_df_all$col_label   <- factor(bin_df_all$col_label, levels = plot_order)
  bin_df_all <- bin_df_all %>% mutate(cluster_num = as.numeric(col_label), x_pos = cluster_num - 0.4 + (bin - 1) / (n_bins - 1) * 0.8,
                                      y_base = row_num_rev, y_top = row_num_rev + score_scaled * 0.85)
  row_labels_ordered <- rev(plot_order)
  y_colors_final <- cluster_colors[row_labels_ordered]

  axis_label_size <- 18     # 1.5 x the earlier 12 pt
  p <- ggplot(bin_df_all) +
    geom_rect(data = clusters_df, aes(xmin = cluster_num - 0.48, xmax = cluster_num + 0.48, ymin = 0.5, ymax = n_total + 0.5, fill = stripe_color), alpha = 0.18, inherit.aes = FALSE) +
    geom_ribbon(aes(x = x_pos, ymin = y_base, ymax = y_top, fill = fill_color, group = interaction(row_label, col_label)), alpha = 0.9) +
    scale_fill_identity() +

add_module_scores_named <- function(obj, module_list, prefix) {
  for (mod_name in names(module_list)) {
    genes_present <- intersect(module_list[[mod_name]], rownames(obj))
    full_name <- paste0(prefix, "_", mod_name)
    if (length(genes_present) == 0) { message("No genes present for ", full_name, " -- skipped"); next }
    obj <- AddModuleScore(obj, features = list(genes_present), name = "TmpModScore")
    obj[[full_name]] <- obj$TmpModScore1
    obj$TmpModScore1 <- NULL
  }
  obj
}
report_missing_genes <- function(module_list, axis_group_name, all_object_genes) {
  rows <- list()
  for (mod_name in names(module_list)) {
    missing_here <- setdiff(module_list[[mod_name]], all_object_genes)
    if (length(missing_here) > 0) rows[[mod_name]] <- data.frame(axis_group = axis_group_name, module = mod_name, gene = missing_here, stringsAsFactors = FALSE)
  }
  if (length(rows) > 0) bind_rows(rows) else data.frame(axis_group = character(0), module = character(0), gene = character(0))
}

# ---- Score once: lineage-restricted combined object per lineage; full individual objects ----
score_lineage <- function(lineage_pattern, axes) {
  lin <- subset(combined, subset = grepl(lineage_pattern, cell_identity))
  for (grp in names(axes)) lin <- add_module_scores_named(lin, axes[[grp]], grp)
  mod_cols <- intersect(unlist(lapply(names(axes), function(grp) paste0(grp, "_", names(axes[[grp]])))), colnames(lin@meta.data))
  df <- lin@meta.data[, c("dataset", "cell_identity", "implant_region", mod_cols)]
  df$dataset <- as.character(df$dataset); df$cell_identity <- as.character(df$cell_identity)
  list(obj = lin, df = df, mod_cols = mod_cols)
}

run_axis_violin_common <- function(combined_lineage_obj, modules, module_labels, conditions, comparison_tag, out_dir) {
  sub_obj <- subset(combined_lineage_obj, subset = dataset %in% conditions)
  sub_obj$dataset <- factor(sub_obj$dataset, levels = conditions)
  shared_identities <- intersect(unique(sub_obj$cell_identity[sub_obj$dataset == conditions[1]]), unique(sub_obj$cell_identity[sub_obj$dataset == conditions[2]]))
  if (length(shared_identities) == 0) { message("No shared identities for ", comparison_tag, " -- skipped"); return(invisible(NULL)) }
  sub_obj <- subset(sub_obj, subset = cell_identity %in% shared_identities)
  identity_levels <- intersect(master_identity_order, shared_identities)
  sub_obj$cell_identity <- factor(sub_obj$cell_identity, levels = identity_levels)
  modules_present <- intersect(modules, colnames(sub_obj@meta.data))
  if (length(modules_present) == 0) { message("No scored modules present for ", comparison_tag); return(invisible(NULL)) }
  plot_df <- FetchData(sub_obj, vars = c("cell_identity", "dataset", modules_present))
  plot_df_long <- plot_df %>%
    pivot_longer(cols = all_of(modules_present), names_to = "module", values_to = "score") %>%
    mutate(module_label = module_labels[module], dataset = factor(dataset, levels = conditions), cell_identity = factor(cell_identity, levels = identity_levels))
  colors_use <- dataset_colors[conditions]
  axis_th <- theme(axis.text.x = element_text(angle = 45, hjust = 1, size = 12, face = "bold"), axis.text.y = element_text(size = 12, face = "bold"))
  for (mod in modules_present) {
    df_mod <- plot_df_long %>% filter(module == mod, !is.na(score))
    stat_df <- tryCatch({
      df_mod %>% group_by(cell_identity) %>% filter(n_distinct(dataset) == 2) %>%
        wilcox_test(score ~ dataset) %>% add_significance("p") %>% add_xy_position(x = "cell_identity", dodge = 0.85)
    }, error = function(e) data.frame())
    p <- ggplot(df_mod, aes(x = cell_identity, y = score, fill = dataset)) +
      geom_violin(position = position_dodge(width = 0.85), trim = TRUE, alpha = 0.75, scale = "width") +
      geom_boxplot(position = position_dodge(width = 0.85), width = 0.15, outlier.shape = NA, alpha = 0.9) +
      scale_fill_manual(values = colors_use) + theme_classic() + axis_th + theme(plot.title = element_text(size = 16, face = "bold")) +
      labs(title = paste0(module_labels[mod], ": ", conditions[1], " vs ", conditions[2], " (shared cell identities)"), x = NULL, y = "Module score", fill = "Object")
    if (nrow(stat_df) > 0) p <- p + stat_pvalue_manual(stat_df, label = "p.signif", hide.ns = TRUE, tip.length = 0.01)
    ggsave(paste0(out_dir, "Violin_Boxplot_", mod, "_", comparison_tag, ".jpeg"), p, width = 11, height = 7, dpi = 300)
    ggsave(paste0(out_dir, "Violin_Boxplot_", mod, "_", comparison_tag, ".pdf"),  p, width = 11, height = 7)
  }
  n_mod <- length(modules_present)
  p_all <- ggplot(plot_df_long, aes(x = cell_identity, y = score, fill = dataset)) +
    geom_violin(position = position_dodge(width = 0.85), trim = TRUE, alpha = 0.75, scale = "width") +
    geom_boxplot(position = position_dodge(width = 0.85), width = 0.15, outlier.shape = NA, alpha = 0.9) +
    facet_wrap(~ module_label, scales = "free_y", ncol = 3) +
    scale_fill_manual(values = colors_use) + theme_classic() + axis_th +
    theme(strip.text = element_text(size = 12, face = "bold"), plot.title = element_text(size = 17, face = "bold")) +
    labs(title = paste0("All modules: ", conditions[1], " vs ", conditions[2]), x = NULL, y = "Module score", fill = "Object")
  ggsave(paste0(out_dir, "Violin_Boxplot_ALL_", comparison_tag, ".jpeg"), p_all, width = 6 * min(3, n_mod), height = 5 * ceiling(n_mod / 3), dpi = 300)
  ggsave(paste0(out_dir, "Violin_Boxplot_ALL_", comparison_tag, ".pdf"),  p_all, width = 6 * min(3, n_mod), height = 5 * ceiling(n_mod / 3))
}

make_axis_spatial_shared_scale <- function(obj1, obj1_name, pt1, sq1, obj2, obj2_name, pt2, sq2, mod_col, mod_label, out_dir) {
  if (!(mod_col %in% colnames(obj1@meta.data)) || !(mod_col %in% colnames(obj2@meta.data))) return(invisible(NULL))
  shared_range <- range(c(FetchData(obj1, vars = mod_col)[, 1], FetchData(obj2, vars = mod_col)[, 1]), na.rm = TRUE)
  mk <- function(obj, nm, pt, sq) {
    p <- SpatialFeaturePlot(obj, features = mod_col, pt.size.factor = pt) +
      scale_fill_gradientn(colors = spatial_colors_100, limits = shared_range, name = mod_label) +
      ggtitle(paste0(mod_label, " (", nm, ")")) +
      theme(plot.title = element_text(size = 13, face = "bold"), legend.title = element_text(face = "bold"), legend.text = element_text(face = "bold"))
    if (sq) p <- p + theme(aspect.ratio = 1)
    p
  }
  p_spatial <- mk(obj1, obj1_name, pt1, sq1) + mk(obj2, obj2_name, pt2, sq2) +
    plot_annotation(title = paste0(mod_label, " (shared color scale): ", obj1_name, " vs ", obj2_name))
  ggsave(paste0(out_dir, "spatial_", mod_col, "_", obj1_name, "_vs_", obj2_name, ".png"), p_spatial, width = 10, height = 5, dpi = 300)
}

get_module_label <- function(module, module_short) {
  lbl <- unname(module_display_labels[module]); fb <- is.na(lbl); lbl[fb] <- cap_label(module_short[fb]); lbl
}
status_from <- function(diff, significant, prob_sup) {
  dir <- ifelse(abs(diff) < unchanged_cutoff, sign(prob_sup - 0.5), sign(diff))
  ifelse(!significant, "ns", ifelse(dir > 0, "higher", ifelse(dir < 0, "lower", "ns")))
}

compute_axis_stats <- function(score_df, mod_cols, axes, ref, other, lineage_tag) {
  d <- score_df[score_df$implant_region == "Implant" & score_df$dataset %in% c(ref, other), ]
  n_tab <- table(d$dataset, d$cell_identity)
  ok_ids <- colnames(n_tab)[n_tab[ref, ] >= dumbbell_min_spots & n_tab[other, ] >= dumbbell_min_spots]
  shared <- intersect(master_identity_order, ok_ids)
  if (length(shared) == 0) { message("No shared implant identities for ", lineage_tag, ": ", ref, " vs ", other); return(NULL) }
  map_df <- bind_rows(lapply(names(axes), function(g) data.frame(axis_group = g, module = paste0(g, "_", names(axes[[g]])), module_short = names(axes[[g]]), stringsAsFactors = FALSE)))
  map_df <- map_df[map_df$module %in% mod_cols, ]
  rows <- vector("list", length(shared) * nrow(map_df)); k <- 0
  for (id in shared) {
    d_id <- d[d$cell_identity == id, ]
    for (m in map_df$module) {
      x_ref <- d_id[[m]][d_id$dataset == ref];   x_ref <- x_ref[!is.na(x_ref)]
      x_oth <- d_id[[m]][d_id$dataset == other]; x_oth <- x_oth[!is.na(x_oth)]
      wt <- suppressWarnings(wilcox.test(x_oth, x_ref, exact = FALSE))
      k <- k + 1
      rows[[k]] <- data.frame(identity = id, module = m, n_ref = length(x_ref), n_other = length(x_oth),
                              median_ref = median(x_ref), median_other = median(x_oth), mean_ref = mean(x_ref), mean_other = mean(x_oth),
                              W = unname(wt$statistic), p_raw = wt$p.value, stringsAsFactors = FALSE)
    }
  }
  bind_rows(rows) %>% left_join(map_df, by = "module") %>%
    group_by(identity) %>% mutate(p_adj_BH = p.adjust(p_raw, method = "BH")) %>% ungroup() %>%
    mutate(lineage = lineage_tag, reference_object = ref, compared_object = other, comparison = paste0(other, "_minus_", ref),
           identity_type = ifelse(grepl("&", identity), "mixed", "pure"), module_label = get_module_label(module, module_short),
           median_diff = median_other - median_ref, mean_diff = mean_other - mean_ref, prob_superiority = W / (n_other * n_ref),
           significant = !is.na(p_adj_BH) & p_adj_BH < sig_cutoff,
           stars = case_when(is.na(p_adj_BH) ~ "ns", p_adj_BH < 1e-4 ~ "****", p_adj_BH < 1e-3 ~ "***", p_adj_BH < 1e-2 ~ "**", p_adj_BH < sig_cutoff ~ "*", TRUE ~ "ns"),
           status_median = status_from(median_diff, significant, prob_superiority),
           status_mean   = status_from(mean_diff,   significant, prob_superiority),
           unchanged_but_significant_median = significant & abs(median_diff) < unchanged_cutoff,
           unchanged_but_significant_mean   = significant & abs(mean_diff)   < unchanged_cutoff) %>%
    select(lineage, comparison, reference_object, compared_object, identity, identity_type, axis_group, module, module_label,
           n_ref, n_other, median_ref, median_other, median_diff, mean_ref, mean_other, mean_diff, W, prob_superiority, p_raw, p_adj_BH,
           stars, significant, status_median, status_mean, unchanged_but_significant_median, unchanged_but_significant_mean)
}

plot_axis_dumbbell <- function(st, axes, ref, other, out_prefix, stat = c("median", "mean")) {
  stat <- match.arg(stat)
  st <- st %>% mutate(plot_diff = if (stat == "median") median_diff else mean_diff,
                      status = status_from(plot_diff, significant, prob_superiority),
                      unch_sig = significant & abs(plot_diff) < unchanged_cutoff)
  ids <- intersect(master_identity_order, unique(st$identity)); n_id <- length(ids)
  shape_vals <- setNames(c(21, 23, 24, 22, 25)[seq_len(n_id)], ids)
  other_lab <- gsub("_", " ", other); ref_lab <- gsub("_", " ", ref)
  ord <- st %>% group_by(axis_group, module, module_label) %>% summarise(mean_val = mean(plot_diff), .groups = "drop")
  lay_list <- list(); cursor <- 0
  for (g in names(axes)) {
    og <- ord[ord$axis_group == g, ]; og <- og[order(-og$mean_val), ]
    if (nrow(og) == 0) next
    if (cursor > 0) cursor <- cursor + 0.6
    sec <- if (g %in% names(section_titles)) unname(section_titles[g]) else cap_label(g)
    lay_list[[length(lay_list) + 1]] <- data.frame(kind = "header", module = NA_character_, label = sec, y = -cursor, idx = 0, stringsAsFactors = FALSE)
    cursor <- cursor + 1
    for (i in seq_len(nrow(og))) {
      lay_list[[length(lay_list) + 1]] <- data.frame(kind = "module", module = og$module[i], label = og$module_label[i], y = -cursor, idx = i, stringsAsFactors = FALSE)
      cursor <- cursor + 1
    }
  }
  lay <- bind_rows(lay_list); mods <- lay[lay$kind == "module", ]; hdrs <- lay[lay$kind == "header", ]
  row_h <- 0.30; lab_w <- max(nchar(lay$label)) * 0.09 + 0.3; data_w <- 5.2; gap <- 0.3; star_w <- 0.65
  x0 <- lab_w + gap
  xr <- range(c(0, st$plot_diff)); xr <- xr + c(-1, 1) * max(0.12 * diff(xr), 0.02)
  scl <- data_w / diff(xr); xmap <- function(v) x0 + (v - xr[1]) * scl
  brks <- pretty(xr, n = 5); brks <- brks[brks >= xr[1] & brks <= xr[2]]
  digits <- if (all(abs(brks - round(brks, 1)) < 1e-9)) 1 else 2
  tick_lab <- ifelse(brks == 0, formatC(0, format = "f", digits = digits), sub("-", "\u2212", sprintf(paste0("%+.", digits, "f"), brks)))
  star_x0 <- x0 + data_w + 0.45; total_w <- star_x0 + n_id * star_w + 0.1
  y_top <- 0.6; y_grid_bot <- min(lay$y) - 0.5; y_bot <- y_grid_bot - 2.4
  pts <- st %>% inner_join(mods[, c("module", "y", "idx")], by = "module") %>%
    mutate(x = xmap(plot_diff), identity = factor(identity, levels = ids), status = factor(status, levels = c("higher", "lower", "ns")),
           star_x = star_x0 + (as.integer(identity) - 0.5) * star_w)
  ann <- pts %>% filter(unch_sig) %>%
    mutate(txt = if (n_id == 1) paste0(stat, " unchanged; distribution ", ifelse(status == "higher", "higher", "lower")) else paste0(as.character(identity), " ", stat, " unchanged")) %>%
    group_by(y) %>% summarise(txt = paste(txt, collapse = "; "), .groups = "drop")
  axis_title <- paste0("Difference in ", stat, " module score (", other_lab, " minus ", ref_lab, "), ", paste(ids, collapse = " / "), " spots")
  p <- ggplot() +
    geom_rect(data = mods[mods$idx %% 2 == 1, ], aes(xmin = 0, xmax = total_w, ymin = y - 0.5, ymax = y + 0.5), fill = "#F2F2F7") +
    geom_segment(data = data.frame(b = brks), aes(x = xmap(b), xend = xmap(b), y = y_grid_bot, yend = 0.4), colour = "#E1E1E8", linewidth = 0.4) +
    annotate("segment", x = xmap(0), xend = xmap(0), y = y_grid_bot, yend = 0.4, colour = "#6B6B7B", linewidth = 0.7) +
    geom_segment(data = pts, aes(x = xmap(0), xend = x, y = y, yend = y), colour = "#CFCFDB", linewidth = 0.7) +
    geom_point(data = pts, aes(x = x, y = y, fill = status, shape = identity), size = 4.4, colour = "white", stroke = 0.5) +
    geom_text(data = hdrs, aes(x = lab_w, y = y, label = label), hjust = 1, fontface = "bold", size = 4.8) +    # section headers right-aligned with module names
    geom_text(data = mods, aes(x = lab_w, y = y, label = label), hjust = 1, size = 4.3) +
    geom_text(data = pts[pts$stars != "ns", ], aes(x = star_x, y = y, label = stars), fontface = "bold", size = 5, vjust = 0.3) +
    geom_text(data = pts[pts$stars == "ns", ], aes(x = star_x, y = y, label = stars), colour = "grey55", size = 3.8) +
    geom_text(data = data.frame(x = xmap(brks), y = y_grid_bot - 0.7, label = tick_lab), aes(x = x, y = y, label = label), size = 4.2, colour = "grey30") +
    annotate("text", x = xmap(mean(xr)), y = y_grid_bot - 1.7, label = axis_title, size = 4.2, colour = "grey30") +
    scale_fill_manual(values = c(higher = "#D64541", lower = "#2F5DA8", ns = "#BCBCC8"), breaks = c("higher", "lower", "ns"),
                      labels = c(paste0("Higher in ", other_lab), paste0("Lower in ", other_lab), "Not significant"), name = NULL, drop = FALSE) +
    scale_shape_manual(values = shape_vals, name = NULL) +
    scale_x_continuous(limits = c(0, total_w), expand = c(0, 0)) + scale_y_continuous(limits = c(y_bot, y_top), expand = c(0, 0)) +
    guides(fill = guide_legend(order = 1, override.aes = list(shape = 21, size = 4.4)),
           shape = if (n_id > 1) guide_legend(order = 2, override.aes = list(fill = "grey70", size = 4.4)) else "none") +
    theme_void(base_size = 12) +
    theme(legend.position = "top", legend.justification = "left", legend.text = element_text(size = 12), plot.margin = margin(8, 8, 8, 8))
  if (n_id > 1) p <- p + geom_text(data = data.frame(x = star_x0 + (seq_len(n_id) - 0.5) * star_w, y = hdrs$y[1], label = ids), aes(x = x, y = y, label = label), fontface = "bold", size = 4.6)
  if (nrow(ann) > 0) p <- p + geom_text(data = ann, aes(x = xmap(0) + 0.3, y = y, label = txt), hjust = 0, fontface = "italic", colour = "grey35", size = 3.5)
  w <- total_w + 0.4; h <- (y_top - y_bot) * row_h + 1.0
  png(paste0(out_prefix, ".png"), width = w * 300, height = h * 300, res = 300, type = "cairo", bg = "white"); print(p); dev.off()
  cairo_pdf(paste0(out_prefix, ".pdf"), width = w, height = h); print(p); dev.off()
}

}
